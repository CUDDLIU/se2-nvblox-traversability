cmake_minimum_required(VERSION 3.8)
project(se2_scan_bridge)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -O3)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
set(deps rclcpp plan_env bspline_opt scan_planner_msgs nav_msgs sensor_msgs geometry_msgs std_msgs tf2 tf2_geometry_msgs visualization_msgs)
foreach(dep ${deps})
  find_package(${dep} REQUIRED)
endforeach()
add_executable(scan_bridge src/bridge.cpp)
# Sources stay inside the project; installation script places a symlink in src/.
get_filename_component(integration_real "${CMAKE_CURRENT_SOURCE_DIR}" REALPATH)
target_include_directories(scan_bridge PRIVATE "${integration_real}/../terrain_variants/deskew")
ament_target_dependencies(scan_bridge ${deps})
target_link_libraries(scan_bridge Eigen3::Eigen)
install(TARGETS scan_bridge DESTINATION lib/${PROJECT_NAME})
ament_package()
